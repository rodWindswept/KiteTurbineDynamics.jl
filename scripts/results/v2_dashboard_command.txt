using KiteTurbineDynamics
# Simulate like interactive_dashboard.jl does for V10
include("scripts/builders_util.jl")
# Try V10 tight
p = KiteTurbineDynamics.params_v6_50kw()

    multibody_ode!(du,u,(sys,p,wf),t)
    u[3N+1:6N].+=DT.*du[3N+1:6N];u[1:3N].+=DT.*u[3N+1:6N]

    u[6N+1:6N+Nr].+=DT.*u[6N+Nr+1:6N+2Nr]
    u[1:3].=0;u[3N+1:3N+3].=0
    s%SE==0&&push!(efs,capture_extended(u,sys,p,t,wf))
end
ef=efs[end]
println("V6 50kW: P=",round(ef.base.P_kw;digits=1),"kW FoS=",round(ef.base.fos_ring;digits=1))
