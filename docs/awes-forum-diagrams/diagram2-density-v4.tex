\documentclass[tikz, border=10pt]{standalone}
\usetikzlibrary{calc}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=0.85]

% Background
\fill[black!1, rounded corners=8pt] (-12.5,-7.5) rectangle (12.5,6.5);
\node[font=\bfseries\Huge, black!80] at (0,5.8) {Why the Density Profile Changed: From Bottom-Biased to Uniform};
\node[font=\large, black!50] at (0,5.1) {At n=3, per-beam compression is high — rings must cluster at the bottom. At n=12, load is distributed.};

% === LEFT: n=3 ===
\begin{scope}[xshift=-7.0cm, yshift=-0.2cm]
  \fill[orange!3, rounded corners=6pt] (-5.0,-7.0) rectangle (5.0,4.5);
  \node[font=\bfseries\Large, orange!60!black] at (0,4.0) {n = 3, $\beta$ = 0.76};
  \node[font=\small, orange!50!black] at (0,3.4) {Heavily bottom-biased};

  % Shaft axis
  \draw[gray!40, line width=1pt] (0,-5.5) -- (0,2.8);

  % Triangle rings — bottom-biased
  \foreach \i in {0,...,9} {
    \pgfmathsetmacro{\t}{\i/9.0}
    \pgfmathsetmacro{\tbias}{\t^(1-0.76)}
    \pgfmathsetmacro{\yy}{2.8 - \tbias*7.5}
    \pgfmathsetmacro{\rr}{0.4 + \tbias*2.2}
    \pgfmathsetmacro{\thick}{2.2 - \tbias*1.3}
    \draw[orange!70!black, line width=\thick pt]
      ({\rr*cos(90)},{\yy+\rr*sin(90)}) --
      ({\rr*cos(210)},{\yy+\rr*sin(210)}) --
      ({\rr*cos(330)},{\yy+\rr*sin(330)}) -- cycle;
  }

  % Ground
  \draw[thick, black!50] (-4.0,-5.0) -- (4.0,-5.0);
  \node[font=\footnotesize, black!50] at (0,-5.4) {ground};

  % Compression annotation
  \draw[->, red!70, line width=2.5pt] (2.9,-4.0) -- (2.9,0.5);
  \node[font=\small\bfseries, red!70, align=center, fill=white, fill opacity=0.8, text opacity=1.0, rounded corners=2pt, inner sep=2pt] at (3.8,-1.5) {Cumulative\\compression\\peaks at\\bottom ring\\$\rightarrow$ rings must\\be tight here};

  % Stats
  \node[font=\footnotesize, align=left, fill=white, draw=gray!20, rounded corners=3pt, inner sep=5pt] at (0,-6.3) {
    Per-beam compression: $N/3$ --- high\\
    Bottom L/r = 0.79 \quad Buckling FoS: 15.7$\times$ at base\\
    \textbf{Why:} Tight spacing resists high per-beam load};
\end{scope}

% === CENTER ARROW ===
\draw[->, line width=4pt, >=stealth, green!50!black] (-1.5,0.0) -- (1.5,0.0)
  node[midway,above,font=\bfseries\Large] {n=3 $\rightarrow$ n=12};

% === RIGHT: n=12 ===
\begin{scope}[xshift=7.0cm, yshift=-0.2cm]
  \fill[blue!2, rounded corners=6pt] (-5.0,-7.0) rectangle (5.0,4.5);
  \node[font=\bfseries\Large, blue!60!black] at (0,4.0) {n = 12, $\beta$ = $-$0.13};
  \node[font=\small, blue!50!black] at (0,3.4) {Nearly uniform spacing};

  % Shaft axis
  \draw[gray!40, line width=1pt] (0,-5.5) -- (0,2.8);

  % Dodecagon rings — uniform spacing
  \foreach \i in {0,...,8} {
    \pgfmathsetmacro{\t}{\i/8.0}
    \pgfmathsetmacro{\tbias}{\t^(1-(-0.13))}
    \pgfmathsetmacro{\yy}{2.8 - \tbias*7.5}
    \pgfmathsetmacro{\rr}{0.8 + \tbias*1.6}
    \draw[blue!50!black, line width=0.55pt]
      ({\rr*cos(15)},{\yy+\rr*sin(15)})
      foreach \a in {45,75,...,345} { -- ({\rr*cos(\a)},{\yy+\rr*sin(\a)}) } -- cycle;
  }

  % Ground
  \draw[thick, black!50] (-4.0,-5.0) -- (4.0,-5.0);
  \node[font=\footnotesize, black!50] at (0,-5.4) {ground};

  % Distributed load annotation
  \draw[->, blue!70, line width=1pt] (2.9,2.0) -- (2.9,-4.0);
  \node[font=\small\bfseries, blue!70, align=center, fill=white, fill opacity=0.8, text opacity=1.0, rounded corners=2pt, inner sep=2pt] at (3.8,-1.5) {Low per-beam\\compression\\everywhere\\$\rightarrow$ no need\\for local\\reinforcement};

  % Stats
  \node[font=\footnotesize, align=left, fill=white, draw=gray!20, rounded corners=3pt, inner sep=5pt] at (0,-6.3) {
    Per-beam compression: $N/12$ --- $4\times$ lower\\
    L/r $\approx$ 3.0 everywhere \quad Buckling FoS: uniform\\
    \textbf{Why:} Low per-beam load means uniform spacing is sufficient};
\end{scope}

\end{tikzpicture}
\end{document}
