% === LEFT: n=3, beta=+0.76 ===
\begin{scope}[xshift=-8.5cm, yshift=-0.1cm]
  \fill[orange!3, rounded corners=6pt] (-5.0,-6.5) rectangle (5.0,4.5);
  \node[font=\bfseries\Large, orange!60!black] at (0,3.9) {$n = 3$, $\beta = +0.76$};
  \node[font=\small, orange!50!black] at (0,3.3) {$\beta{>}0$: rings cluster at bottom};

  \draw[gray!40, line width=1pt] (0,-5.0) -- (0,2.8);

  \foreach \i in {0,...,9} {
    \pgfmathsetmacro{\t}{\i/9.0}
    \pgfmathsetmacro{\tbias}{\t^(1-0.76)}
    \pgfmathsetmacro{\yy}{2.8 - \tbias*7.2}
    \pgfmathsetmacro{\rr}{2.5 - \tbias*2.15}
    \pgfmathsetmacro{\thick}{0.4 + \tbias*1.8}
    \draw[orange!70!black, line width=\thick pt]
      ({\rr*cos(90)},{\yy+\rr*sin(90)}) --
      ({\rr*cos(210)},{\yy+\rr*sin(210)}) --
      ({\rr*cos(330)},{\yy+\rr*sin(330)}) -- cycle;
  }

  \draw[thick, black!50] (-4.0,-4.5) -- (4.0,-4.5);
  \node[font=\footnotesize, black!50] at (0,-4.9) {ground (smallest ring)};

  \node[font=\small\bfseries, orange!70!black, align=center, fill=white, fill opacity=0.9, text opacity=1.0, rounded corners=3pt, inner sep=3pt] at (3.8,1.5) {Large rings\\at top (hub)};
  \node[font=\small\bfseries, orange!70!black, align=center, fill=white, fill opacity=0.9, text opacity=1.0, rounded corners=3pt, inner sep=3pt] at (3.8,-2.5) {Small rings\\tightly packed\\at bottom\\--- high compression};

  \node[font=\footnotesize, align=left, fill=white, draw=gray!20, rounded corners=3pt, inner sep=4pt] at (0,-5.8) {
    Per-beam compression: $N/3$ --- very high\\
    \textbf{$\beta{=}{+}0.76$:} Rings cluster at bottom where cumulative load peaks.};
\end{scope}

% === RIGHT: n=12, beta=-0.13 ===
\begin{scope}[xshift=8.5cm, yshift=-0.1cm]
  \fill[blue!2, rounded corners=6pt] (-5.0,-6.5) rectangle (5.0,4.5);
  \node[font=\bfseries\Large, blue!60!black] at (0,3.9) {$n = 12$, $\beta = -0.13$};
  \node[font=\small, blue!50!black] at (0,3.3) {$\beta{<}0$: mild top-bias --- sign inverted};

  \draw[gray!40, line width=1pt] (0,-5.0) -- (0,2.8);

  \foreach \i in {0,...,8} {
    \pgfmathsetmacro{\t}{\i/8.0}

    \pgfmathsetmacro{\rr}{2.3 - \tbias*1.6}
    \draw[blue!50!black, line width=0.55pt]
      ({\rr*cos(15)},{\yy+\rr*sin(15)})

  }

  \node[font=\small\bfseries, blue!60!black, align=center, fill=white, fill opacity=0.9, text opacity=1.0, rounded corners=3pt, inner sep=3pt] at (4.2,-2.5) {Small rings\\at bottom\\--- nearly uniform\\spacing};

  \node[font=\footnotesize, align=left, fill=white, draw=gray!20, rounded corners=3pt, inner sep=4pt] at (0,-5.8) {

    \textbf{$\beta{=}{-}0.13$:} Compression so low that beam taper (thin near hub)\\
    \quad now dominates --- rings drift slightly toward top where beams are thinnest.};
\end{scope}
